│  GET /api/ipo/price
│  Output: {price, subscription, volume, min_shares}
│
├─ IPO SUBSCRIPTION
│  POST /api/ipo/subscribe
│  Input:  {email, shares}
│  Output: {success, message, amount, id}
│
├─ STATISTICS
│  GET /api/stats
│  Output: {registrations, active}
│
└─ HEALTH CHECK
   GET /health

💾 DATABASE SCHEMA:

TABLE: registrations
  ├─ id (INTEGER, PK)
  ├─ name (TEXT)
  ├─ email (TEXT)
  ├─ phone (TEXT)
  ├─ national_id (TEXT)
  ├─ service (TEXT)
  └─ created_at (DATETIME)

TABLE: ipo_subscriptions
  ├─ id (INTEGER, PK)
  ├─ email (TEXT)
  ├─ shares (INTEGER)
  ├─ amount (REAL)
  └─ created_at (DATETIME)

TABLE: ipo_data
  ├─ id (INTEGER, PK)
  ├─ price (REAL)
  ├─ subscription_percent (REAL)
  ├─ total_shares (INTEGER)
  └─ last_updated (DATETIME)
